# 単一exeとして作成する (例)

dotnet publish McpOracle/McpOracle.csproj -c Release -r win-x64 --self-contained true -p:PublishSingleFile=true -p:IncludeNativeLibrariesForSelfExtract=true -p:PublishTrimmed=false -p:DebugType=None -o C:\gitroot\tools\McpOracle

# MCPサーバとして登録する Claude例
   "mcpServers": {
      "cv-oracle": {
         "command": "C:\gitroot\tools\McpOracle\McpOracle.exe",
         "args": ["Data Source=IPアドレス/DB名;User Id=ユーザ名;Password=パスワード;"]
      }
   }

# MCPサーバとして登録する Codex例

config.toml に追加

[mcp_servers.cv-oracle]
enabled = true
command = 'C:\gitroot\tools\McpOracle\McpOracle.exe'
args = ['Data Source=IPアドレス/DB名;User Id=ユーザ名;Password=パスワード;']
